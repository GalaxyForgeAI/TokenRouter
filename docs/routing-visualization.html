改造后策略接口 + 注册表（可插拔）

// ① 定义接口
type RouteStrategy interface {
  Name() string
  Apply(group []RouteSelection) // 组内排序/权重修正
}

// ② 全局注册表
var registry = map[string]RouteStrategy{}

func RegisterStrategy(s RouteStrategy) {
  registry[s.Name()] = s
}

// ③ 网关分发：switch → 查表
strategy := registry[route.Strategy]
strategy.Apply(group)   // 未知 → 默认 priority_weighted

// ④ 新算法 = 实现接口 + 注册
func init() { RegisterStrategy(LowestLatency{}) }
